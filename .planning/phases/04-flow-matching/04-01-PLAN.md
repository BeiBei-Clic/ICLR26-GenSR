def sample_timestep(batch_size):
    if args.timestep_dist == "uniform":
        return torch.rand(batch_size, device=device)
    else:
        u = torch.randn(batch_size, device=device)
        return torch.sigmoid(args.logit_normal_loc + args.logit_normal_scale * u)

# cola_sft.py:189-196 — LR schedule
def get_lr_multiplier(progress):
    if progress < args.warmup_ratio:
        return (progress + 1e-8) / args.warmup_ratio
    elif progress <= 1.0 - args.warmdown_ratio:
        return 1.0
    else:
        decay = (progress - (1.0 - args.warmdown_ratio)) / args.warmdown_ratio
        return (1 - decay) * 1.0 + decay * args.final_lr_frac

    2. flow_matching_step(dit, prior_mu, post_mu, timestep_dist="logit_normal"):
       - GenSR 极简版 FM step（对照 cola_sft.py:451-534 大幅简化）
       - per D-09: 无序列、无 mask、无 2L trick
       - B = prior_mu.shape[0]
       - t = sample_timestep(B, dist=timestep_dist)  → (B,)

       - loss = ((pred - target) ** 2).mean()  → 标量，per FM-02 无 mask
       - 返回 (loss, loss.detach())
       - per FM-01, FM-02

    3. get_lr_multiplier(progress, warmup_ratio=0.05, warmdown_ratio=0.3, final_lr_frac=0.0):

       ```
       smooth_loss = 0.0
       ema_beta = 0.95  # D-07
       start_step = start_step (从 resume 或 0)
       num_iterations = args.num_iterations

       for step in range(start_step, num_iterations):
           t0 = time.time()
           last_step = step == num_iterations - 1

           # --- Eval ---
           if step == 0 or last_step or (args.eval_every > 0 and step % args.eval_every == 0):
               val_loss = evaluate(dit, eval_iter, args.eval_steps)
               print(f"Step {step:05d} | Val FM loss: {val_loss:.6f}")

           # --- Save ---
           if last_step or (args.save_every > 0 and step > 0 and step % args.save_every == 0):
               save_checkpoint(dit, optimizer, step, val_loss, os.path.join(args.output_dir, f"run"))

           # --- Training step with gradient accumulation ---
           for micro_step in range(args.grad_accum_steps):
               prior_mu, post_mu = next(train_iter)
               prior_mu = prior_mu.cuda()
               post_mu = post_mu.cuda()
               loss, train_loss = flow_matching_step(dit, prior_mu, post_mu, args.timestep_dist)
               (loss / args.grad_accum_steps).backward()  # per Pitfall 5: train_loss 在缩放前获取

           # LR schedule
           progress = step / max(num_iterations - 1, 1)
           lrm = get_lr_multiplier(progress, args.warmup_ratio, args.warmdown_ratio, args.final_lr_frac)
           for group in optimizer.param_groups:
               group["lr"] = group["initial_lr"] * lrm

           # Optimizer step
           torch.nn.utils.clip_grad_norm_(dit.parameters(), 1.0)  # D-06
           optimizer.step()
           optimizer.zero_grad(set_to_none=True)

           dt = time.time() - t0

           # EMA smoothing (D-07)
           smooth_loss = ema_beta * smooth_loss + (1 - ema_beta) * train_loss.item()
           debiased = smooth_loss / (1 - ema_beta ** (step - start_step + 1))

           if step % 10 == 0 or last_step:
               print(f"step {step:05d} | loss: {debiased:.6f} | lr: {lrm * args.learning_rate:.2e} | dt: {dt * 1000:.0f}ms")
       ```
