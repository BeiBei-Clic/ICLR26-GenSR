2. 在 Node 类中添加 compile_to_instructions 方法（在 val 方法之后，约第 235 行后）。
**关键：fallback 机制** -- 对于不在 _OP_MAP 和 _MATH_CONST 中的非变量节点值，尝试 float(val)，如果 ValueError（说明是未支持的操作符如 tan/arctan），返回 None 作为标记：
```python
def compile_to_instructions(self):
    """后序遍历树，生成扁平指令列表 (opcodes, constants, var_dims)。
    返回 None 表示该树包含不支持的操作符，需要 fallback 到 Python val()。
    """
    opcodes, constants, var_dims = [], [], []
    for child in self.children:
        result = child.compile_to_instructions()
        if result is None:
            return None
        oc, co, vd = result
        opcodes.extend(oc); constants.extend(co); var_dims.extend(vd)
    val = self.value
    if val.startswith('x_'):
        opcodes.append(OP_VAR)
        var_dims.append(int(val.split('_')[1]))
    elif val in _OP_MAP:
        opcodes.append(_OP_MAP[val])
    else:
        # 检查是否为已知常量或可转换为浮点数的值
        if val in _MATH_CONST:
            opcodes.append(OP_CONST)
            constants.append(_MATH_CONST[val])
        else:
            opcodes.append(OP_CONST)
            try:
                constants.append(float(val))
            except ValueError:
                # 不支持的操作符（如 tan, arctan, arcsin, arccos 等）
                # 标记整棵树需要 fallback 到 Python val()
                return None
    return opcodes, constants, var_dims
```

fallback 逻辑说明：compile_to_instructions 在遇到未知节点值（如 'tan'、'arctan'）时 float(val) 会 ValueError，此时返回 None。父节点收到子树返回 None 后也返回 None，最终整棵树标记为需要 fallback。这样不会在运行时崩溃，只是对这些树回退到 Python 路径。由于 tan/arctan 的出现概率较低（0.2），绝大多数树仍走 Cython 快速路径。

3. 在 NodeList 类中添加 val_cython 方法（在 val 方法之后，约第 282 行后）。

```python
def val_cython(self, xs, deterministic=True):
    """使用 Cython 栈式求值器计算 NodeList 的值。
    包含不支持操作符的树自动 fallback 到 Python val()。
    deterministic 参数保持接口一致（Cython 路径始终 deterministic）。
    """
    from symbolicregression.envs.cython_eval import eval_tree_flat
    x = np.ascontiguousarray(xs, dtype=np.float64)
    batch_vals = []
    for node in self.nodes:
        compiled = node.compile_to_instructions()
        if compiled is None:
            # fallback 到 Python 路径
            batch_vals.append(np.expand_dims(node.val(np.copy(xs), deterministic=deterministic), -1))
        else:
            oc, co, vd = compiled
            opcodes = np.array(oc, dtype=np.int32)
            constants = np.array(co, dtype=np.float64)
            var_dims = np.array(vd, dtype=np.int32)
            result = eval_tree_flat(x, opcodes, constants, var_dims)
            batch_vals.append(np.expand_dims(result, -1))
    return np.concatenate(batch_vals, axis=-1)
```

```python
"""Cython 求值器性能基准测试。"""
import numpy as np
import time
import sys
sys.path.insert(0, '.')

from symbolicregression.envs.generators import Node, NodeList
from symbolicregression.envs.cython_eval import eval_tree_flat

class FakeParams:
    use_abs = False

def build_tree():
    """构造一棵典型复杂度的表达式树（约 15-20 节点）。"""
    params = FakeParams()
    # mul(sin(x_0), add(x_0, cos(mul(x_0, 2.5))))
    x0 = Node('x_0', params)
    sin_x0 = Node('sin', params, [x0])

    x0b = Node('x_0', params)
    const_25 = Node('2.5', params)
    mul_inner = Node('mul', params, [x0b, const_25])
    cos_inner = Node('cos', params, [mul_inner])
    x0c = Node('x_0', params)
    add_inner = Node('add', params, [x0c, cos_inner])

    root = Node('mul', params, [sin_x0, add_inner])
    return NodeList([root])

def compile_nodelist(nodelist):
    from symbolicregression.envs.generators import OP_VAR, OP_CONST, OP_ADD, OP_SUB, OP_MUL, OP_DIV
    from symbolicregression.envs.generators import OP_SIN, OP_COS, OP_EXP, OP_LOG, OP_SQRT
    from symbolicregression.envs.generators import OP_ABS, OP_POW2, OP_POW3, OP_INV, OP_NEG
    compiled = []
    for node in nodelist.nodes:
        compiled_node = node.compile_to_instructions()
        if compiled_node is None:
            raise RuntimeError("Benchmark tree should not contain unsupported operators")
        oc, co, vd = compiled_node
        compiled.append((
            np.array(oc, dtype=np.int32),
            np.array(co, dtype=np.float64),
            np.array(vd, dtype=np.int32),
        ))
    return compiled

# 构建测试数据
tree = build_tree()
compiled = compile_nodelist(tree)
n_samples = 200
x = np.random.randn(n_samples, 1).astype(np.float64)

# Warmup
for _ in range(100):
    tree.val(x)
for _ in range(100):
    eval_tree_flat(np.ascontiguousarray(x), compiled[0][0], compiled[0][1], compiled[0][2])

# Benchmark Python
n_runs = 5000
t0 = time.perf_counter()
for _ in range(n_runs):
    tree.val(x)
t_python = (time.perf_counter() - t0) / n_runs * 1e6  # us

# Benchmark Cython
t0 = time.perf_counter()
for _ in range(n_runs):
    for opcodes, constants, var_dims in compiled:
        eval_tree_flat(np.ascontiguousarray(x), opcodes, constants, var_dims)
t_cython = (time.perf_counter() - t0) / n_runs * 1e6  # us

print(f"Python Node.val(): {t_python:.1f} us/call")
print(f"Cython eval_tree_flat: {t_cython:.1f} us/call")
print(f"Speedup: {t_python / t_cython:.1f}x")
```
