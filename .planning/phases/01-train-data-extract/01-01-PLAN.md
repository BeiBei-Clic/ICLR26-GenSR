    **LatentPairDataset(IterableDataset)** 核心实现：

    构造函数 `__init__(self, params, batch_size=16, device="cuda")`：
    - 接收 params（argparse.Namespace）、batch_size、device
    - 调用 `build_env(params)` 创建 env
    - 调用 `build_modules(env, params)` 创建 modules
    - 调用 `reload_model(modules, modules_to_load=["cvae", "data_encoder", "token_embed"], path="weights/checkpoint.pth", requires_grad=False)` 加载冻结权重
    - 设置 `env.rng = np.random.RandomState()` （per D-02 不固定种子）
    - 保存 embedder_f = modules["data_encoder"], embedder_e = modules["token_embed"], vae_model = modules["cvae"]
    - 设置 `vae_model.eval()` 并冻结
    - 设置 `self.env = env`

    `__iter__` 方法，`while True` 循环：
    ```python
    def __iter__(self):
        while True:
            with torch.no_grad():
                # 1. 生成 batch_size 个样本
                samples, _ = self.env.gen_expr(train=True)
                # samples["X_to_fit"] 是 List[np.ndarray]，长度为 batch_size
                # 需要将 batch_size 个样本展开为 x_to_fit, y_to_fit 列表

                x_to_fit = samples["X_to_fit"]   # List[np.ndarray], 每个 (n_points, dim)
                y_to_fit = samples["Y_to_fit"]   # List[np.ndarray], 每个 (n_points, 1)

                # 2. 构造 x1（复刻 trainer_vae.py:862-871）
                x1 = []
                for seq_id in range(len(x_to_fit)):
                    x1.append([])
                    for seq_l in range(len(x_to_fit[seq_id])):
                        x1[seq_id].append([x_to_fit[seq_id][seq_l], y_to_fit[seq_id][seq_l]])
                x1, len1 = self.embedder_f(x1)

                # 3. 构造 x2（复刻 trainer_vae.py:880-886）
                x2, len2 = self.env.batch_equations(
                    self.env.word_to_idx(samples["tree_encoded"], float_input=False)
                )
                x2, len2 = to_cuda(x2, len2)
                x2_e = self.embedder_e(x2.transpose(0, 1)).transpose(0, 1)

                # 4. CVAE forward（复刻 trainer_vae.py:888）
                prior_mu, _, post_mu, _, _, _, _ = self.vae_model(x1, x2_e, len1, len2, mode="train")

                # 5. 逐个 yield (prior_mu[i], post_mu[i])
                for i in range(prior_mu.shape[0]):
                    yield prior_mu[i], post_mu[i]
    ```

    **重要细节：**
    - `env.gen_expr(train=True)` 默认每次生成的样本数量由 FunctionEnvironment 内部决定（通常等于 batch_size 参数），但实际数量可能不等于 batch_size。因此 `__iter__` 的 while True 循环中，每次 gen_expr 可能产出不同数量的样本。逐个 yield 即可，DataLoader 会自动 collate 到 batch_size。
    - 使用 `with torch.no_grad():` 包裹所有 CVAE 前向传播，避免显存泄漏。
    - `import symbolicregression.utils` 并设置 `symbolicregression.utils.CUDA = True`，否则 `to_cuda` 不会生效。
    - `x1, len1 = self.embedder_f(x1)` 返回的已经是 CUDA 张量（embedder_f 内部有 .cuda()），不需要再 to_cuda。
    - 路径 "weights/checkpoint.pth" 使用相对于项目根目录的路径。

    **create_latent_dataloader 工厂函数：**
    ```python
    def create_latent_dataloader(params, batch_size=16, num_workers=0, device="cuda"):
        dataset = LatentPairDataset(params, batch_size=batch_size, device=device)
        return DataLoader(
            dataset,
            batch_size=batch_size,
            num_workers=num_workers,  # 必须 0，CUDA 不支持多进程
            pin_memory=False,
        )
    ```
    注意 num_workers 默认为 0 且强制为 0（CUDA 多进程限制）。

    功能：用 LatentPairDataset 生成 N 个 batch，打印统计信息，供 Phase 2 数据质量验证使用。

    核心逻辑：
    ```python
    """快速验证 LatentPairDataset 数据生成质量。用法: python -m dit_train.data.verify_dataset"""
    import sys, os
    sys.path.insert(0, os.path.join(os.path.dirname(__file__), "..", ".."))
    import torch
    import numpy as np
    from parsers import get_parser
    from dit_train.data.latent_dataset import create_latent_dataloader

    def main():
        params = get_parser().parse_args([])
        params.device = "cuda"
        batch_size = 8
        n_batches = 5

        dataloader = create_latent_dataloader(params, batch_size=batch_size)

        prior_mus = []
        post_mus = []
        for i, (prior_mu, post_mu) in enumerate(dataloader):
            if i >= n_batches:
                break
            prior_mus.append(prior_mu.cpu().numpy())
            post_mus.append(post_mu.cpu().numpy())
            print(f"Batch {i}: prior_mu {prior_mu.shape}, post_mu {post_mu.shape}")
            print(f"  prior_mu mean={prior_mu.mean().item():.4f}, std={prior_mu.std().item():.4f}")
            print(f"  post_mu  mean={post_mu.mean().item():.4f}, std={post_mu.std().item():.4f}")
            print(f"  diff norm={torch.norm(prior_mu - post_mu, dim=1).mean().item():.4f}")

        prior_all = np.concatenate(prior_mus, axis=0)
        post_all = np.concatenate(post_mus, axis=0)
        print(f"\n=== 总计 {len(prior_all)} 个样本 ===")
        print(f"prior_mu: mean={prior_all.mean():.4f}, std={prior_all.std():.4f}, range=[{prior_all.min():.4f}, {prior_all.max():.4f}]")
        print(f"post_mu:  mean={post_all.mean():.4f}, std={post_all.std():.4f}, range=[{post_all.min():.4f}, {post_all.max():.4f}]")
        print(f"mean diff norm: {np.linalg.norm(prior_all - post_all, axis=1).mean():.4f}")

    if __name__ == "__main__":
        main()
    ```

    脚本可以直接运行：`python -m dit_train.data.verify_dataset`
    不保存任何文件到磁盘（per D-01）。
